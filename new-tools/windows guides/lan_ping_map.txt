# Script PowerShell para pingar todos os IPs da sub-rede 192.168.4.0/24

$subnet = "192.168.4"
1..254 | ForEach-Object {
    $ip = "$subnet.$_"
    Write-Host "Testando $ip..." -ForegroundColor Cyan
    if (Test-Connection -ComputerName $ip -Count 1 -Quiet) {
        Write-Host "$ip está respondendo." -ForegroundColor Green
    } else {
        Write-Host "$ip não respondeu." -ForegroundColor Red
    }
}
